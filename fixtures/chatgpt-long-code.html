class NumberAnalysisService:

def __init__(self):
self._analyzers = [
NaturalNumberAnalyzer(),
WholeNumberAnalyzer(),
SignAnalyzer(),
EvenOddAnalyzer(),
PrimeNumberAnalyzer(),
CompositeNumberAnalyzer(),
PalindromeNumberAnalyzer(),
ArmstrongNumberAnalyzer(),
PerfectNumberAnalyzer(),
]

def analyze_all(self, value):
number = Number(value)
results = {}

for analyzer in self._analyzers:
results[analyzer.name] = analyzer.analyze(number)

return results

def print_report(self, value):
number = Number(value)

print("\n" + "=" * 60)
print(f"NUMBER ANALYSIS REPORT: {value}")
print("=" * 60)

print(f"Absolute value : {number.absolute_value()}")
print(f"Digit count : {number.digit_count()}")
print(f"Digit sum : {number.digit_sum()}")
print(f"Reversed number : {number.reverse_digits()}")
print(f"First digit : {number.first_digit()}")
print(f"Last digit : {number.last_digit()}")
print(f"Digit frequency : {number.digit_frequency()}")

print("\nNumber properties:")

for analyzer in self._analyzers:
result = analyzer.analyze(number)
print(f"{analyzer.name:<23}: {result}")

armstrong = ArmstrongNumberAnalyzer()

print(
"\nArmstrong digit-power sum:",
armstrong.calculate_armstrong_sum(number)
)

if value > 0:
divisors = DivisorFinder().find(value)
proper = PerfectNumberAnalyzer.proper_divisors(value)

print("Positive divisors :", divisors)
print("Number of divisors :", len(divisors))
print("Proper divisor sum :", sum(proper))
else:
print("Positive divisors : only available for n > 0")

print("=" * 60)

return self.analyze_all(value)
